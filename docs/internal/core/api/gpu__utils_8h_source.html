61inline wgpu::ComputePipeline
62cached_pipeline(const std::string& shader_id, const std::string& label) {
63 // Cache key includes the raw device pointer so that a device recreation
64 // (which yields a different WGPUDevice handle) automatically invalidates
65 // stale entries. A mutex guards the map against concurrent access.
66 using CacheKey = std::pair<WGPUDevice, std::string>;
67 struct PairHash {
68 size_t operator()(const CacheKey& k) const {
69 auto h1 = std::hash<const void*> {}(static_cast<const void*>(k.first));
70 auto h2 = std::hash<std::string> {}(k.second);
71 return h1 ^ (h2 << 1);
72 }
73 };
74
75 static std::mutex mtx;
76 static std::unordered_map<CacheKey, wgpu::ComputePipeline, PairHash> cache;
77
78 WGPUDevice raw_device = GPU::getClassInstance().get_device().Get();
79 CacheKey key {raw_device, shader_id};
80
81 std::lock_guard<std::mutex> lock(mtx);
82 auto it = cache.find(key);
83 if (it != cache.end()) {
84 return it->second;
85 }
86 wgpu::ComputePipeline pipeline = GPU::getClassInstance().createPipeline(shader_id, label);
87 cache[key] = pipeline;
88 return pipeline;
89}

104inline const void* map_and_wait(wgpu::Buffer& buffer, size_t size) {
105 struct MapState {
106 bool done = false;
107 bool success = false;
108 };
109
110 MapState state;
111
112 buffer.MapAsync(
113 wgpu::MapMode::Read, 0, size, wgpu::CallbackMode::AllowProcessEvents,
114 [](wgpu::MapAsyncStatus status, wgpu::StringView /*msg*/, void* userdata) {
115 auto* s = static_cast<MapState*>(userdata);
116 s->done = true;
117 s->success = (status == wgpu::MapAsyncStatus::Success);
118 },
119 static_cast<void*>(&state)
120 );
121
122 while (!state.done) {
123 GPU::getClassInstance().get_instance().ProcessEvents();
124#if defined(__EMSCRIPTEN__)
125 emscripten_sleep(10);
126#endif
127 }
128
129 if (!state.success) {
130 throw std::runtime_error("gpu::map_and_wait: MapAsync failed");
131 }
132
133 const void* mapped = buffer.GetConstMappedRange(0, size);
134 if (!mapped) {
135 throw std::runtime_error("gpu::map_and_wait: GetConstMappedRange returned null");
136 }
137 return mapped;
138}

150template <typename T> class Buffer {
151 static_assert(std::is_trivially_copyable_v<T>, "Buffer<T>: T must be trivially copyable");
152
153 public:
157 Buffer(size_t count, wgpu::BufferUsage usage)
158 : count_(count)
159 , byte_size_(count * sizeof(T)) {
160 wgpu::BufferDescriptor desc {};
161 desc.size = byte_size_;
162 desc.usage = usage;
163 buffer_ = GPU::getClassInstance().get_device().CreateBuffer(&desc);
164 }
165
170 void write(const T* data, size_t count, size_t offset_elements = 0) {
171 GPU::getClassInstance().get_queue().WriteBuffer(
172 buffer_, offset_elements * sizeof(T), data, count * sizeof(T)
173 );
174 }
175
177 void write(const std::vector<T>& data) {
178 write(data.data(), data.size());
179 }
180
185 std::vector<T> read() {
186 // Create staging buffer
187 wgpu::BufferDescriptor staging_desc {};
188 staging_desc.size = byte_size_;
189 staging_desc.usage = wgpu::BufferUsage::MapRead | wgpu::BufferUsage::CopyDst;
190 wgpu::Buffer staging = GPU::getClassInstance().get_device().CreateBuffer(&staging_desc);
191
192 // Copy GPU buffer → staging
193 wgpu::CommandEncoder encoder = GPU::getClassInstance().get_device().CreateCommandEncoder();
194 encoder.CopyBufferToBuffer(buffer_, 0, staging, 0, byte_size_);
195 wgpu::CommandBuffer commands = encoder.Finish();
196 GPU::getClassInstance().get_queue().Submit(1, &commands);
197
198 // Map and wait
199 const void* mapped = map_and_wait(staging, byte_size_);
200
201 // Copy to vector
202 std::vector<T> result(count_);
203 std::memcpy(result.data(), mapped, byte_size_);
204
205 staging.Unmap();
206 staging.Destroy();
207 return result;
208 }
209
211 const wgpu::Buffer& handle() const {
212 return buffer_;
213 }
214
216 size_t byte_size() const {
217 return byte_size_;
218 }
219
221 size_t size() const {
222 return count_;
223 }
224
225 // Non-copyable
226 Buffer(const Buffer&) = delete;
227 Buffer& operator=(const Buffer&) = delete;
228
230 Buffer(Buffer&& other) noexcept
231 : buffer_(std::move(other.buffer_))
232 , count_(other.count_)
233 , byte_size_(other.byte_size_) {
234 other.buffer_ = nullptr;
235 other.count_ = 0;
236 other.byte_size_ = 0;
237 }
238
240 Buffer& operator=(Buffer&& other) noexcept {
241 if (this != &other) {
242 if (buffer_) {
243 buffer_.Destroy();
244 }
245 buffer_ = std::move(other.buffer_);
246 count_ = other.count_;
247 byte_size_ = other.byte_size_;
248 other.buffer_ = nullptr;
249 other.count_ = 0;
250 other.byte_size_ = 0;
251 }
252 return *this;
253 }
254
256 ~Buffer() {
257 if (buffer_) {
258 buffer_.Destroy();
259 }
260 }
261
262 private:
263 wgpu::Buffer buffer_ {nullptr};
264 size_t count_ {0};
265 size_t byte_size_ {0};
266};

298 void write(const void* data, size_t data_size, uint32_t bytes_per_pixel) {
299 wgpu::TexelCopyTextureInfo dst {};
300 dst.texture = texture_;
301 wgpu::TexelCopyBufferLayout layout {};
302 layout.offset = 0;
303 layout.bytesPerRow = width_ * bytes_per_pixel;
304 layout.rowsPerImage = height_;
305 wgpu::Extent3D extent {width_, height_, 1};
306 GPU::getClassInstance().get_queue().WriteTexture(&dst, data, data_size, &layout, &extent);
307 }
308
315 std::vector<uint8_t> read(uint32_t bytes_per_pixel) {
316 uint32_t aligned_bpr = GPU::getAlignedBytesPerRow(width_, bytes_per_pixel);
317 uint32_t staging_size = aligned_bpr * height_;
318
319 // Create staging buffer
320 wgpu::BufferDescriptor staging_desc {};
321 staging_desc.size = staging_size;
322 staging_desc.usage = wgpu::BufferUsage::MapRead | wgpu::BufferUsage::CopyDst;
323 wgpu::Buffer staging = GPU::getClassInstance().get_device().CreateBuffer(&staging_desc);
324
325 // Encode copy
326 wgpu::CommandEncoder encoder = GPU::getClassInstance().get_device().CreateCommandEncoder();
327
328 wgpu::TexelCopyTextureInfo src {};
329 src.texture = texture_;
330 wgpu::TexelCopyBufferInfo dst {};
331 dst.buffer = staging;
332 dst.layout.bytesPerRow = aligned_bpr;
333 dst.layout.rowsPerImage = height_;
334 wgpu::Extent3D extent {width_, height_, 1};
335 encoder.CopyTextureToBuffer(&src, &dst, &extent);
336
337 wgpu::CommandBuffer commands = encoder.Finish();
338 GPU::getClassInstance().get_queue().Submit(1, &commands);
339
340 // Map and wait
341 const auto* mapped = static_cast<const uint8_t*>(map_and_wait(staging, staging_size));
342
343 // Strip row alignment padding
344 uint32_t tight_bpr = width_ * bytes_per_pixel;
345 std::vector<uint8_t> result(tight_bpr * height_);
346
347 for (uint32_t y = 0; y < height_; ++y) {
348 std::memcpy(result.data() + y * tight_bpr, mapped + y * aligned_bpr, tight_bpr);
349 }
350
351 staging.Unmap();
352 staging.Destroy();
353 return result;
354 }

478inline wgpu::BindGroup make_bind_group(
479 const wgpu::ComputePipeline& pipeline, uint32_t group_index,
480 const std::vector<BindEntry>& entries
481) {
482 std::vector<wgpu::BindGroupEntry> wgpu_entries;
483 wgpu_entries.reserve(entries.size());
484
485 for (const auto& e : entries) {
486 wgpu::BindGroupEntry wge {};
487 wge.binding = e.binding;
488 if (e.texture_view) {
489 wge.textureView = e.texture_view;
490 }
491 if (e.buffer) {
492 wge.buffer = e.buffer;
493 wge.size = e.buffer_size;
494 wge.offset = e.buffer_offset;
495 }
496 wgpu_entries.push_back(wge);
497 }
498
499 wgpu::BindGroupDescriptor desc {};
500 desc.layout = pipeline.GetBindGroupLayout(group_index);
501 desc.entryCount = static_cast<uint32_t>(wgpu_entries.size());
502 desc.entries = wgpu_entries.data();
503
504 return GPU::getClassInstance().get_device().CreateBindGroup(&desc);
505}

521inline void encode_pass(
522 wgpu::CommandEncoder& encoder, const wgpu::ComputePipeline& pipeline,
523 const wgpu::BindGroup& bind_group, uint32_t groups_x, uint32_t groups_y = 1,
524 uint32_t groups_z = 1
525) {
526 wgpu::ComputePassEncoder pass = encoder.BeginComputePass();
527 pass.SetPipeline(pipeline);
528 pass.SetBindGroup(0, bind_group);
529 pass.DispatchWorkgroups(groups_x, groups_y, groups_z);
530 pass.End();
531}
532
542inline void dispatch(
543 const wgpu::ComputePipeline& pipeline, const wgpu::BindGroup& bind_group, uint32_t groups_x,
544 uint32_t groups_y = 1, uint32_t groups_z = 1
545) {
546 wgpu::CommandEncoder encoder = GPU::getClassInstance().get_device().CreateCommandEncoder();
547 encode_pass(encoder, pipeline, bind_group, groups_x, groups_y, groups_z);
548 wgpu::CommandBuffer commands = encoder.Finish();
549 GPU::getClassInstance().get_queue().Submit(1, &commands);
550}
