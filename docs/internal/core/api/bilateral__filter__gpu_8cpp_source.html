46void bilateral_filter_gpu(
47 uint8_t* image, size_t width, size_t height, double sigma_spatial, double sigma_range,
48 uint8_t color_space
49) {
50 if (sigma_spatial <= 0.0 || sigma_range <= 0.0 || width <= 0 || height <= 0)
51 return;
52 if (color_space != gpu::COLOR_SPACE_CIELAB && color_space != gpu::COLOR_SPACE_RGB)
53 return;
54
55 const uint32_t w = static_cast<uint32_t>(width);
56 const uint32_t h = static_cast<uint32_t>(height);
57 constexpr uint32_t bpp = 4; // RGBA8 = 4 bytes per pixel
58
59 IMG2NUM_LOG_INFO("begin wgpu portion");
60
61 // 1. Create textures (RAII — destroyed automatically on scope exit)
62 gpu::Texture inputTexture(
63 w, h, wgpu::TextureFormat::RGBA8Unorm,
64 wgpu::TextureUsage::TextureBinding | wgpu::TextureUsage::CopyDst
65 );
66
67 IMG2NUM_LOG_INFO("upload texture");
68 inputTexture.write(image, bpp * width * height, bpp);
69
70 IMG2NUM_LOG_INFO("create output texture");
71 gpu::Texture outputTexture(
72 w, h, wgpu::TextureFormat::RGBA8Unorm,
73 wgpu::TextureUsage::StorageBinding | wgpu::TextureUsage::CopySrc
74 );
75
76 // Intermediate textures for CIELAB color-space filtering
77 gpu::Texture texLabRaw(
78 w, h, wgpu::TextureFormat::RGBA32Float,
79 wgpu::TextureUsage::StorageBinding | wgpu::TextureUsage::TextureBinding
80 );
81 gpu::Texture texLabFiltered(
82 w, h, wgpu::TextureFormat::RGBA32Float,
83 wgpu::TextureUsage::StorageBinding | wgpu::TextureUsage::TextureBinding
84 );
85
86 IMG2NUM_LOG_INFO("create buffer");
87 // 2. Create uniform buffer with filter parameters
88 gpu::Buffer<FilterParams> paramBuffer(
89 1, wgpu::BufferUsage::Uniform | wgpu::BufferUsage::CopyDst
90 );
91 FilterParams params = {
92 static_cast<float>(sigma_spatial), static_cast<float>(sigma_range), 0.0f, 0.0f
93 };
94 paramBuffer.write(&params, 1);
95
96 // 3. Create pipelines
97 wgpu::ComputePipeline pipeline;
98 wgpu::ComputePipeline pipelineRGB2LAB;
99 wgpu::ComputePipeline pipelineLAB2RGB;
100
101 switch (color_space) {
102 case gpu::COLOR_SPACE_RGB:
103 pipeline = gpu::cached_pipeline("bilateral_filter_rgb", "BilateralFilterShader");
104 break;
105 case gpu::COLOR_SPACE_CIELAB:
106 pipeline = gpu::cached_pipeline("bilateral_filter_lab", "BilateralFilterShader");
107 pipelineRGB2LAB = gpu::cached_pipeline("rgb2cielab", "rgb2lab");
108 pipelineLAB2RGB = gpu::cached_pipeline("cielab2rgb", "lab2rgb");
109 break;
110 }
111
112 // 4. Create bind groups
113 wgpu::BindGroup bindGroup;
114 wgpu::BindGroup bindGroupRGB2LAB;
115 wgpu::BindGroup bindGroupLAB2RGB;
116
117 switch (color_space) {
118 case gpu::COLOR_SPACE_RGB:
119 bindGroup = gpu::make_bind_group(
120 pipeline, 0,
121 {
122 gpu::BindEntry::texture(0, inputTexture),
123 gpu::BindEntry::texture(1, outputTexture),
124 gpu::BindEntry::buffer_entry(2, paramBuffer),
125 }
126 );
127 break;
128 case gpu::COLOR_SPACE_CIELAB:
129 bindGroup = gpu::make_bind_group(
130 pipeline, 0,
131 {
132 gpu::BindEntry::texture(0, texLabRaw),
133 gpu::BindEntry::texture(1, texLabFiltered),
134 gpu::BindEntry::buffer_entry(2, paramBuffer),
135 }
136 );
137 bindGroupRGB2LAB = gpu::make_bind_group(
138 pipelineRGB2LAB, 0,
139 {
140 gpu::BindEntry::texture(0, inputTexture),
141 gpu::BindEntry::texture(1, texLabRaw),
142 }
143 );
144 bindGroupLAB2RGB = gpu::make_bind_group(
145 pipelineLAB2RGB, 0,
146 {
147 gpu::BindEntry::texture(0, texLabFiltered),
148 gpu::BindEntry::texture(1, outputTexture),
149 }
150 );
151 break;
152 }
153
154 // 5. Dispatch compute passes
155 uint32_t wgX = gpu::workgroup_count(w);
156 uint32_t wgY = gpu::workgroup_count(h);
157
158 wgpu::CommandEncoder encoder = GPU::getClassInstance().get_device().CreateCommandEncoder();
159
160 if (color_space == gpu::COLOR_SPACE_CIELAB) {
161 gpu::encode_pass(encoder, pipelineRGB2LAB, bindGroupRGB2LAB, wgX, wgY);
162 }
163
164 gpu::encode_pass(encoder, pipeline, bindGroup, wgX, wgY);
165
166 if (color_space == gpu::COLOR_SPACE_CIELAB) {
167 gpu::encode_pass(encoder, pipelineLAB2RGB, bindGroupLAB2RGB, wgX, wgY);
168 }
169
170 wgpu::CommandBuffer commands = encoder.Finish();
171 GPU::getClassInstance().get_queue().Submit(1, &commands);
172 IMG2NUM_LOG_INFO("queue submit");
173
174 // 6. Readback — handles staging buffer, MapAsync, alignment stripping automatically
175 std::vector<uint8_t> result = outputTexture.read(bpp);
176 std::memcpy(image, result.data(), result.size());
177 IMG2NUM_LOG_INFO("done memcpy");
178
179 // All textures and buffers are destroyed automatically by RAII destructors
180#if defined(__EMSCRIPTEN__)
181 emscripten_sleep(50);
182#endif
183}
